1. 两数之和（哈希表）  
   时间 O(n) 空间 O(n)

数组中存在两个数字等于 target

```js
// 返回的是数组的索引
const twoSum = (nums, target) => {
  const map = new Map();
  for (let i = 0; i < nums.length; i++) {
    const com = target - nums[i];
    if (map.has(com)) return [map.get(com), i];
    map.set(nums[i], i);
  }
};
// 空间换时间
var twoSum = function (nums, target) {
  const arr = [];
  for (let i = 0; i < nums.length; i++) {
    const com = target - nums[i];
    if (arr[com] !== undefined) {
      return [arr[com], i];
    } else {
      arr[nums[i]] = i;
    }
  }
};
```

2. 最长无重复子串（滑动窗口）  
   时间 O(n) 空间 O(字符集)

map 记录每个字符最后出现的位置，一旦后续查询到前面有重复，就更新该值

```js
const lengthOfLongestSubstring = (s) => {
  let left = 0,
    max = 0,
    map = new Map();
  for (let right = 0; right < s.length; right++) {
    if (map.has(s[right])) left = Math.max(left, map.get(s[right]) + 1);
    map.set(s[right], right);
    max = Math.max(max, right - left + 1);
  }
  return max;
};
```

非递归版

```js
function dfs(root) {
  if (!root) return [];
  const stack = [root],
    res = [];
  while (stack.length) {
    const node = stack.pop();
    res.push(node.val);
    // 子节点倒序入栈保证左→右顺序
    for (let i = (node.children || []).length - 1; i >= 0; i--) {
      stack.push(node.children[i]);
    }
  }
  return res;
}
```

4. 二叉树广度优先遍历（队列）  
   时间 O(n) 空间 O(w)

```js
function bfs(root) {
  if (!root) return [];
  const q = [root],
    res = [];
  while (q.length) {
    const node = q.shift();
    res.push(node.val);
    (node.children || []).forEach((child) => q.push(child));
  }
  return res;
}
```

```js
// 斐波那契数列 f(n-1) + f(n-2) 起始值不同
const climbStairs = (n) => {
  let [a, b] = [1, 1]; // a = f(n-2)  b = f(n-1)
  for (let i = 2; i <= n; i++) [a, b] = [b, a + b];
  return b;
};
```

// 求数组中的最大差值
const maxProfit = (prices) => {
  let min = Infinity,
    max = 0;

    min = Math.min(min, p);
    max = Math.max(max, p - min);

6.1 求最大容水面积

给定一个长度为 n 的整数数组 height 。有 n 条垂线，第 i 条线的两个端点是 (i, 0) 和 (i, height[i]) 。

找出其中的两条线，使得它们与 x 轴共同构成的容器可以容纳最多的水。

返回容器可以储存的最大水量。

```js
const maxArea = (height) => {
  let left = 0,
    right = height.length - 1,
    max = 0;
  while (left < right) {
    const minHeight = Math.min(height[left], height[right]);
    const area = (right - left) * minHeight;
    max = Math.max(max, area);
    // 矮的指针往中间移动
    if (height[left] < height[right]) left++;
    else right--;
  }
  return max;
};

const maxArea = (height) => {
  let max = 0;
  for (let i = 0, j = height.length - 1; i < j; ) {
    const minHeight = Math.min(height[i], height[j]);
    const area = (j - i) * minHeight;
    max = Math.max(max, area);
    if (height[i] < height[j]) i++;
    else j--;
  }
  return max;
};
```

7. 手写 Promise.all  
   并发控制 + 顺序保证

```js
Promise.myAll = function (promises) {
  return new Promise((resolve, reject) => {
    const res = [],
      total = promises.length;
    let count = 0;
    promises.forEach((p, i) =>
      Promise.resolve(p)
        .then((val) => {
          res[i] = val;
          if (++count === total) resolve(res);
        })
        .catch(reject)
    );
  });
};
```

8. 有效括号（栈）  
   时间 O(n) 空间 O(n)

```js
const isValid = (s) => {
  const map = { "(": ")", "[": "]", "{": "}" },
    stack = [];
  for (const c of s) {
    if (map[c]) stack.push(map[c]);
    else if (stack.pop() !== c) return false;
  }
  return !stack.length;
};
```

10. LRU 缓存（Map + 双向链表）  
    get/put 均 O(1)

```js
class LRU {
  constructor(cap) {
    this.cap = cap;
    this.cache = new Map();
  }
  get(key) {
    if (!this.cache.has(key)) return -1;
    const val = this.cache.get(key);
    this.cache.delete(key); // 提到最新
    this.cache.set(key, val);
    return val;
  }
  put(key, val) {
    if (this.cache.has(key)) this.cache.delete(key);
    else if (this.cache.size === this.cap) {
      const first = this.cache.keys().next().value; // 最老
      this.cache.delete(first);
    }
    this.cache.set(key, val);
  }
}
```
